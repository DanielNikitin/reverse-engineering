bool world_to_screen(const Vec3& w, Vec2& s, const float m[16],
                     int sw, int sh) {
    float cw = m[12]*w.x + m[13]*w.y + m[14]*w.z + m[15];
    if (cw < 0.001f) return false;
    float sx = m[0]*w.x + m[1]*w.y + m[2]*w.z + m[3];
    float sy = m[4]*w.x + m[5]*w.y + m[6]*w.z + m[7];
    s.x = (sw * 0.5f) * (1.0f + sx / cw);
    s.y = (sh * 0.5f) * (1.0f - sy / cw);
    return true;
}
